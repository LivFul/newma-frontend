## Review Focus

1. **Reduced-motion users** — with `prefers-reduced-motion: reduce` every motion duration token must be `0ms`, otherwise the hero (P4) will animate for users who opted out (Task 2 tests the media-query block zeros each duration token).
2. **Hand-edited generated client** — a developer edits `src/lib/api/generated/schema.d.ts`; `api:check` must fail because regeneration produces a diff (Task 5 test runs the script after a mutation and expects exit 1).
3. **Spec substitution** — the pinned SHA-256 no longer matches the fetched spec; `api:check` must fail rather than silently regenerate (Task 5 test with a tampered lock).
4. **Flag leak** — `NEXT_PUBLIC_DEMO_MODE` referenced in a marketing page; the guard must fail (Task 6 test with a fixture tree).
5. **Dialog focus and labelling** — a Dialog without an accessible title is an axe violation; the primitive must require `title` (Task 3 test renders Dialog and asserts zero axe violations and `role="dialog"` with `aria-labelledby`).

- [x] **Step 4: Implement `src/lib/a11y/contrast.ts`**

```ts
// WCAG 2.x relative luminance and contrast ratio.
const channel = (c: number): number => {
  const s = c / 255;
  return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
};

export const expandHex = (hex: string): string => {
  const h = hex.trim().replace("#", "");
  return h.length === 3
    ? h
        .split("")
        .map((ch) => ch + ch)
        .join("")
    : h;
};

export const luminance = (hex: string): number => {
  const h = expandHex(hex);
  const [r, g, b] = [0, 2, 4].map((i) => parseInt(h.slice(i, i + 2), 16));
  return 0.2126 * channel(r) + 0.7152 * channel(g) + 0.0722 * channel(b);
};

export const contrastRatio = (fgHex: string, bgHex: string): number => {
  const [l1, l2] = [luminance(fgHex), luminance(bgHex)].sort((a, b) => b - a);
  return (l1 + 0.05) / (l2 + 0.05);
};

export const parseCssVars = (css: string): Record<string, string> =>
  Object.fromEntries(
    Array.from(css.matchAll(/(--[a-z0-9-]+)\s*:\s*([^;]+);/gi), (m) => [m[1], m[2].trim()]),
  );
```

import axe from "axe-core";
import { expect } from "vitest";

export async function expectNoAxeViolations(container: Element): Promise<void> {
  const results = await axe.run(container, {
    rules: { "color-contrast": { enabled: false } }, // jsdom has no layout; contrast is covered by tokens.test.ts
  });
  const serious = results.violations.filter(
    (v) => v.impact === "serious" || v.impact === "critical",
  );
  expect(serious.map((v) => `${v.id}: ${v.help}`)).toEqual([]);
  expect(results.violations.map((v) => v.id)).toEqual([]);
}

```js
#!/usr/bin/env node
// Contract check (IP §5.6): verify the pinned spec hash, regenerate the typed client, fail on drift.
// Usage: node scripts/api-check.mjs [--update] [--no-typecheck]
import { createHash } from "node:crypto";
import { execFileSync } from "node:child_process";
import { readFileSync, writeFileSync, existsSync } from "node:fs";
import path from "node:path";
import openapiTS, { astToString } from "openapi-typescript";

const cwd = process.cwd();
const repoRoot = process.env.API_CHECK_REPO_ROOT ?? cwd;
const args = new Set(process.argv.slice(2));
const lockPath = path.join(cwd, "api/openapi.lock");
const outPath = path.join(cwd, "src/lib/api/generated/schema.d.ts");

const fail = (msg) => {
  process.stderr.write(`api:check: ${msg}\n`);
  process.exit(1);
};

async function loadSpec(source) {
  if (/^https?:\/\//.test(source)) {
    const res = await fetch(source);
    if (!res.ok) fail(`fetch ${source} -> ${res.status}`);
    return await res.text();
  }
  const p = path.resolve(repoRoot, source);
  if (!existsSync(p)) fail(`spec not found at ${p}`);
  return readFileSync(p, "utf8");
}

const lock = JSON.parse(readFileSync(lockPath, "utf8"));
const spec = await loadSpec(lock.source);
const sha = createHash("sha256").update(spec).digest("hex");
if (args.has("--update")) {
  writeFileSync(lockPath, JSON.stringify({ ...lock, sha256: sha }, null, 2) + "\n");
} else if (sha !== lock.sha256) {
  fail(
    `sha256 mismatch for ${lock.source}: lock ${lock.sha256} vs spec ${sha}. Run pnpm api:update after reviewing the contract change.`,
  );
}
const versionMatch = spec.match(/^\s*version:\s*['"]?([^'"\n]+)['"]?\s*$/m);
if (!versionMatch || versionMatch[1].trim() !== lock.version) {
  fail(`spec version ${versionMatch?.[1]} != lock version ${lock.version}`);
}

const header = `// GENERATED by scripts/api-check.mjs from ${lock.source} (api-v${lock.version}). Do not edit.\n`;
const generated = header + astToString(await openapiTS(spec));
writeFileSync(outPath, generated);
const diff = execFileSync("git", ["status", "--porcelain", "--", path.relative(cwd, outPath)], {
  cwd,
  encoding: "utf8",
});
if (diff.trim() && !args.has("--update")) {
  fail("generated client is stale; run pnpm api:update and commit src/lib/api/generated");
}
if (!args.has("--no-typecheck"))
  execFileSync("pnpm", ["exec", "tsc", "--noEmit"], { cwd, stdio: "inherit" });
process.stdout.write(`api:check: ok (api-v${lock.version}, sha256 ${sha.slice(0, 12)}…)\n`);
```
